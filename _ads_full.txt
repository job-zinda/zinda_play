  15| // Categories of recently watched videos, sent with ad requests for interest targeting.
  16| class Interests {
  17|   static const _storage = FlutterSecureStorage();
  18|   static const _key = 'recent_categories';
  19|   static List<String>? _cache;
  20| 
  21|   static Future<void>? _pending;
  22| 
  23|   static Future<List<String>> read() async {
  24|     await _pending;
  25|     if (_cache != null) return _cache!;
  26|     try {
  27|       final raw = await _storage.read(key: _key);
  28|       _cache = raw == null
  29|           ? []
  30|           : (jsonDecode(raw) as List).map((e) => '$e').toList();
  31|     } catch (_) {
  32|       _cache = [];
  33|     }
  34|     return _cache!;
  35|   }
  36| 
  37|   static Future<void> record(dynamic categoryId) =>
  38|       _pending = _record(categoryId);
  39| 
  40|   static Future<void> _record(dynamic categoryId) async {
  41|     final id = '${categoryId ?? ''}';
  42|     if (id.isEmpty || id == 'null') return;
  43|     final previous = _cache ?? await _load();
  44|     final list = [id, ...previous.where((e) => e != id)].take(8).toList();
  45|     _cache = list;
  46|     try {
  47|       await _storage.write(key: _key, value: jsonEncode(list));
  48|     } catch (_) {}
  49|   }
  50| 
  51|   static Future<List<String>> _load() async {
  52|     try {
  53|       final raw = await _storage.read(key: _key);
  54|       return raw == null
  55|           ? []
  56|           : (jsonDecode(raw) as List).map((e) => '$e').toList();
  57|     } catch (_) {
  58|       return [];
  59|     }
  60|   }
  61| }
  62| 
  63| class AdsApi {
  64|   static final Map<String, int> _videoCounts = {};
  65|   static void markDelivered(String id) => _videoCounts[id] = 0;
  66| 
  67|   static List<Map<String, dynamic>> dueVideos(List<Map<String, dynamic>> ads) {
  68|     return ads.where((ad) {
  69|       final id = '${ad['id']}';
  70|       final count = (_videoCounts[id] ?? 0) + 1;
  71|       _videoCounts[id] = count;
  72|       return count >= ((ad['frequency'] as num?)?.toInt() ?? 1);
  73|     }).toList();
  74|   }
  75| 
  76|   static Future<bool> claim(Map<String, dynamic> ad, {String? ticket}) async {
  77|     final result = await Api.instance.post(
  78|       '/ads/${ad['id']}/impression',
  79|       ticket == null ? null : {'ticket': ticket},
  80|     );
  81|     return (result as Map)['accepted'] == true;
  82|   }
  83| 
  84|   static Future<List<Map<String, dynamic>>> fetch(String placement) async {
  85|     try {
  86|       final interests = Uri.encodeQueryComponent(
  87|         (await Interests.read()).join(','),
  88|       );
  89|       final data = await Api.instance.get(
  90|         '/ads?placement=$placement&interests=$interests',
  91|       );
  92|       return (data as List)
  93|           .map((e) => Map<String, dynamic>.from(e as Map))
  94|           .toList();
  95|     } catch (error) {
  96|       debugPrint('Ad loading failed: $error');
  97|       return [];
  98|     }
  99|   }
 100| 
 101|   static Future<void> track(String id, String event) async {
 102|     try {
 103|       await Api.instance.post('/ads/$id/$event');
 104|     } catch (error) {
 105|       debugPrint('Ad tracking failed: $error');
 106|     }
 107|   }
 108| 
 109|   static Future<void> open(Map<String, dynamic> ad, {String? url}) async {
 110|     url ??= '${ad['targetUrl'] ?? ''}';
 111|     if (url.isEmpty) return;
 112|     track('${ad['id']}', 'click');
 113|     if (!await launchUrl(
 114|       Uri.parse(url),
 115|       mode: LaunchMode.externalApplication,
 116|     )) {
 117|       throw ApiException('The advertiser link could not be opened.');
 118|     }
 119|   }
 120| 
 121|   static Map<String, dynamic> pick(List<Map<String, dynamic>> ads) =>
 122|       ads[Random().nextInt(ads.length)];
 123| }

 278|   Future<void> _start() async {
 279|     if (_started) return;
 280|     _started = true;
 281|     _idleSeconds = 0;
 282|     _loadWatchdog = Timer.periodic(const Duration(seconds: 1), (timer) {
 283|       final value = _video?.value;
 284|       if (!_canPlay ||
 285|           (!_loading &&
 286|               value?.isPlaying == false &&
 287|               value?.isBuffering == false)) {
 288|         _idleSeconds = 0;
 289|         return;
 290|       }
 291|       if (++_idleSeconds >= 15) {
 292|         timer.cancel();
 293|         _video?.pause();
 294|         if (mounted && !_done) {
 295|           setState(
 296|             () => _error = 'The ad is taking too long to load. Check your connection or continue to your video.',
 297|           );
 298|         }
 299|       }
 300|     });
 301|     final ads = AdsApi.dueVideos(await AdsApi.fetch(widget.placement));
 302|     if (!mounted || _done) return;
 303|     if (ads.isEmpty) return _finish();
 304|     final ad = AdsApi.pick(ads);
 305|     final controller = VideoPlayerController.networkUrl(
 306|       Uri.parse(assetUrl('${ad['mediaUrl']}')),
 307|     );
 308|     _preparing = controller;
 309|     try {
 310|       // Media preparation and the watch ticket do not depend on one another.
 311|       await Future.wait([
 312|         controller.initialize().timeout(const Duration(seconds: 15)),
 313|         if (ad['creatorVideoId'] != null)
 314|           Api.instance.post('/ads/${ad['id']}/view-start').then((data) {
 315|             _ticket = Map<String, dynamic>.from(data as Map);
 316|           }),
 317|       ]);
 318|     } catch (error) {
 319|       debugPrint('Video ad initialization failed: $error');
 320|       await _discardPreparing(controller);
 321|       if (mounted && !_done) {
 322|         setState(
 323|           () => _error = 'Could not load this ad. Check your connection or continue to your video.',
 324|         );
 325|       }
 326|       return;
 327|     }
 328|     if (!mounted || _done) {
 329|       await _discardPreparing(controller);
 330|       return;
 331|     }
 332|     try {
 333|       if (!await AdsApi.claim(ad, ticket: _ticket?['ticket'] as String?)) {
 334|         await _discardPreparing(controller);
 335|         if (mounted) _finish();
 336|         return;
 337|       }
 338|       AdsApi.markDelivered('${ad['id']}');
 339|     } catch (error) {
 340|       debugPrint('Video ad delivery failed: $error');
 341|       await _discardPreparing(controller);
 342|       if (mounted) _finish();
 343|       return;
 344|     }
 345|     if (!mounted || _done) {
 346|       await _discardPreparing(controller);
 347|       return;
 348|     }
 349|     _preparing = null;
 350|     controller.addListener(_tick);
 351|     setState(() {
 352|       _ad = ad;
 353|       _video = controller;
 354|       _loading = false;
 355|       _error = null;
 356|     });
 357|     _idleSeconds = 0;
 358|     try {
 359|       if (_canPlay) await controller.play();
 360|     } catch (error) {
 361|       if (mounted) setState(() => _error = 'Could not play this ad: $error');
 362|     }
 363|   }

 365|   Future<void> _discardPreparing(VideoPlayerController controller) async {
 366|     if (_preparing != controller) return;
 367|     _preparing = null;
 368|     await controller.dispose();
 369|   }

 371|   void _tick() {
 372|     final v = _video;
 373|     if (v == null || !mounted || _done) return;
 374|     final value = v.value;
 375|     final position = value.position.inMilliseconds;
 376|     final delta = position - _lastPosition;
 377|     if (delta > 0) _idleSeconds = 0;
 378|     _lastPosition = position;
 379|     final ended =
 380|         value.duration > Duration.zero && value.position >= value.duration;
 381|     if (_canPlay &&
 382|         !value.isBuffering &&
 383|         (value.isPlaying || ended) &&
 384|         delta > 0 &&
 385|         delta < 1500) {
 386|       _watchedMs += delta;
 387|     }
 388|     if (value.hasError) {
 389|       debugPrint('Video ad playback failed: ${value.errorDescription}');
 390|       _finish();
 391|       return;
 392|     }
 393|     if (ended) {
 394|       _complete();
 395|       return;
 396|     }
 397|     setState(() {});
 398|   }

 400|   Future<void> _complete() async {
 401|     if (_completing) return;
 402|     _completing = true;
 403|     _loadWatchdog?.cancel();
 404|     final ticket = _ticket;
 405|     final minimum = (ticket?['minimumWatchMs'] as num?)?.toInt();
 406|     if (ticket != null && minimum != null && _watchedMs >= minimum - 250) {
 407|       try {
 408|         await Api.instance.post('/ads/${_ad!['id']}/view', {
 409|           'ticket': ticket['ticket'],
 410|         });
 411|       } catch (error) {
 412|         debugPrint('Creator ad earning tracking failed: $error');
 413|       }
 414|     }
 415|     if (mounted) {
 416|       if ('${_ad!['targetUrl'] ?? ''}'.isNotEmpty ||
 417|           '${_ad!['productUrl'] ?? ''}'.isNotEmpty) {
 418|         setState(() => _ended = true);
 419|       } else {
 420|         _finish();
 421|       }
 422|     }
 423|   }
