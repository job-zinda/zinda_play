13	class _AdPlayer extends VideoPlayerPlatform {
14	  bool initializeAutomatically = true;
15	  int creations = 0;
16	  int disposals = 0;
17	  late final StreamController<VideoEvent> events = StreamController<VideoEvent>(
18	    onListen: () {
19	      if (initializeAutomatically) initialize();
20	    },
21	  );
22	  void initialize() => events.add(
23	    VideoEvent(
24	      eventType: VideoEventType.initialized,
25	      duration: const Duration(seconds: 3),
26	      size: const Size(720, 1280),
27	    ),
28	  );
29	  Duration position = Duration.zero;
30	  bool playing = false;
31	  @override
32	  Future<void> init() async {}
33	  @override
34	  Future<int?> createWithOptions(VideoCreationOptions options) async {
35	    creations++;
36	    return 1;
37	  }
38	
39	  @override
40	  Stream<VideoEvent> videoEventsFor(int playerId) => events.stream;
41	  @override
42	  Future<void> dispose(int playerId) async {
43	    disposals++;
44	    await events.close();
45	  }
46	
47	  @override
48	  Future<void> play(int playerId) async {
49	    playing = true;
50	  }
51	
52	  @override
53	  Future<void> pause(int playerId) async {
54	    playing = false;
55	  }
56	
57	  @override
58	  Future<void> setVolume(int playerId, double volume) async {}
59	  @override
60	  Future<void> setLooping(int playerId, bool looping) async {}
61	  @override
62	  Future<void> setPlaybackSpeed(int playerId, double speed) async {}
63	  @override
64	  Future<void> seekTo(int playerId, Duration position) async {}
65	  @override
66	  Future<Duration> getPosition(int playerId) async => position;
67	  @override
68	  Widget buildViewWithOptions(VideoViewOptions options) =>
69	      const ColoredBox(color: Colors.black);

300	      'creator ad ${skip ? "skip does not earn" : "full watch posts one earning ticket"}',
301	      (tester) async {
302	        final player = _AdPlayer();
303	        VideoPlayerPlatform.instance = player;
304	        final calls = <String>[];
305	        final adId = skip ? 'skip-ad' : 'full-ad';
306	        final client = MockClient((request) async {
307	          final path = request.url.path;
308	          calls.add('${request.method} $path');
309	          Object body;
310	          if (path == '/api/ads') {
311	            body = [
312	              {
313	                'id': adId,
314	                'type': 'video',
315	                'title': 'Full watch offer',
316	                'targetUrl': 'https://example.com/business',
317	                'productUrl': 'https://example.com/product',
318	                'creatorVideoId': 'simply',
319	                'mediaUrl': 'https://example.b-cdn.net/video/playlist.m3u8',
320	                'frequency': 1,
321	                'skippable': skip,
322	                'skipAfterSeconds': 1,
323	              },
324	            ];
325	          } else if (path.endsWith('/impression')) {
326	            expect(jsonDecode(request.body)['ticket'], 'full-watch-ticket');
327	            expect(
328	              calls.indexOf('POST /api/ads/$adId/view-start'),
329	              lessThan(calls.length - 1),
330	            );
331	            body = {'accepted': true};
332	          } else if (path.endsWith('/view-start')) {
333	            body = {'ticket': 'full-watch-ticket', 'minimumWatchMs': 3000};
334	          } else if (path.endsWith('/view')) {
335	            expect(jsonDecode(request.body)['ticket'], 'full-watch-ticket');
336	            body = {'counted': true};
337	          } else {
338	            return http.Response(
339	              jsonEncode({'error': 'Unexpected request'}),
340	              404,
341	            );
342	          }
343	          return http.Response(jsonEncode(body), 200);
344	        });
345	        await http.runWithClient(() async {
346	          await tester.pumpWidget(
347	            MaterialApp(
348	              home: Scaffold(
349	                body: PlayerGate(builder: (_) => const Text('Main video')),
350	              ),
351	            ),
352	          );
353	          await tester.pumpAndSettle();
354	          expect(player.playing, isTrue);
355	          expect(calls, contains('POST /api/ads/$adId/view-start'));
356	          expect(find.text('Visit advertiser'), findsOneWidget);
357	          final steps = skip ? 4 : 12;
358	          for (int step = 1; step <= steps; step++) {
359	            player.position = Duration(milliseconds: step * 250);
360	            await tester.pump(const Duration(milliseconds: 250));
361	            await tester.pump();
362	            expect(find.text('Visit advertiser'), findsOneWidget);
363	          }
364	          if (skip) {
365	            await tester.binding.setSurfaceSize(const Size(320, 640));
366	            await tester.pump();
367	            final visit = tester.getRect(find.text('Visit advertiser'));
368	            final skipButton = tester.getRect(find.text('Skip ad â€º'));
369	            expect((visit.center.dy - skipButton.center.dy).abs(), lessThan(1));
370	            expect(visit.right, lessThan(skipButton.left));
371	            expect(skipButton.bottom, greaterThan(560));
372	            await tester.binding.setSurfaceSize(null);
373	            await tester.pump();
374	            await tester.tap(find.text('Skip ad â€º'));
375	          } else {
376	            player.events.add(VideoEvent(eventType: VideoEventType.completed));
377	            await tester.pumpAndSettle();
378	            expect(find.text('Visit advertiser'), findsOneWidget);
379	            expect(find.text('View product'), findsOneWidget);
380	            await tester.tap(find.text('Skip ad â€º'));
381	          }
382	          await tester.pumpAndSettle();
383	          expect(find.text('Main video'), findsOneWidget);
384	          expect(
385	            calls.where((call) => call == 'POST /api/ads/$adId/view'),
386	            hasLength(skip ? 0 : 1),
387	          );
388	          await tester.pumpWidget(const SizedBox.shrink());
389	          await tester.pump();
390	        }, () => client);
391	        expect(tester.takeException(), isNull);
392	      },
393	    );
