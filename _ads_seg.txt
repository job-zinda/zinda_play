278:   Future<void> _start() async {
279:     if (_started) return;
280:     _started = true;
281:     _idleSeconds = 0;
282:     _loadWatchdog = Timer.periodic(const Duration(seconds: 1), (timer) {
283:       final value = _video?.value;
284:       if (!_canPlay ||
285:           (!_loading &&
286:               value?.isPlaying == false &&
287:               value?.isBuffering == false)) {
288:         _idleSeconds = 0;
289:         return;
290:       }
291:       if (++_idleSeconds >= 15) {
292:         timer.cancel();
293:         _video?.pause();
294:         if (mounted && !_done) {
295:           setState(
296:             () => _error = 'The ad is taking too long to load. Check your connection or continue to your video.',
297:           );
298:         }
299:       }
300:     });
301:     final ads = AdsApi.dueVideos(await AdsApi.fetch(widget.placement));
302:     if (!mounted || _done) return;
303:     if (ads.isEmpty) return _finish();
304:     final ad = AdsApi.pick(ads);
305:     final controller = VideoPlayerController.networkUrl(
306:       Uri.parse(assetUrl('${ad['mediaUrl']}')),
307:     );
308:     _preparing = controller;
309:     try {
310:       // Media preparation and the watch ticket do not depend on one another.
311:       await Future.wait([
312:         controller.initialize().timeout(const Duration(seconds: 15)),
313:         if (ad['creatorVideoId'] != null)
314:           Api.instance.post('/ads/${ad['id']}/view-start').then((data) {
315:             _ticket = Map<String, dynamic>.from(data as Map);
316:           }),
317:       ]);
318:     } catch (error) {
319:       debugPrint('Video ad initialization failed: $error');
320:       await _discardPreparing(controller);
321:       if (mounted && !_done) {
322:         setState(
323:           () => _error = 'Could not load this ad. Check your connection or continue to your video.',
324:         );
325:       }
326:       return;
327:     }
328:     if (!mounted || _done) {
329:       await _discardPreparing(controller);
330:       return;
331:     }
332:     try {
333:       if (!await AdsApi.claim(ad, ticket: _ticket?['ticket'] as String?)) {
334:         await _discardPreparing(controller);
335:         if (mounted) _finish();
336:         return;
337:       }
338:       AdsApi.markDelivered('${ad['id']}');
339:     } catch (error) {
340:       debugPrint('Video ad delivery failed: $error');
341:       await _discardPreparing(controller);
342:       if (mounted) _finish();
343:       return;
344:     }
345:     if (!mounted || _done) {
346:       await _discardPreparing(controller);
347:       return;
348:     }
349:     _preparing = null;
350:     controller.addListener(_tick);
351:     setState(() {
352:       _ad = ad;
353:       _video = controller;
354:       _loading = false;
355:       _error = null;
356:     });
357:     _idleSeconds = 0;
358:     try {
359:       if (_canPlay) await controller.play();
360:     } catch (error) {
361:       if (mounted) setState(() => _error = 'Could not play this ad: $error');
362:     }
363:   }

365:   Future<void> _discardPreparing(VideoPlayerController controller) async {
366:     if (_preparing != controller) return;
367:     _preparing = null;
368:     await controller.dispose();
369:   }

371:   void _tick() {
372:     final v = _video;
373:     if (v == null || !mounted || _done) return;
374:     final value = v.value;
375:     final position = value.position.inMilliseconds;
376:     final delta = position - _lastPosition;
377:     if (delta > 0) _idleSeconds = 0;
378:     _lastPosition = position;
379:     final ended =
380:         value.duration > Duration.zero && value.position >= value.duration;
381:     if (_canPlay &&
382:         !value.isBuffering &&
383:         (value.isPlaying || ended) &&
384:         delta > 0 &&
385:         delta < 1500) {
386:       _watchedMs += delta;
387:     }
388:     if (value.hasError) {
389:       debugPrint('Video ad playback failed: ${value.errorDescription}');
390:       _finish();
391:       return;
392:     }
393:     if (ended) {
394:       _complete();
395:       return;
396:     }
397:     setState(() {});
398:   }
